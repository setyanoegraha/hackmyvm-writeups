msg = [0, 0, 0, 153, 82, 76, 202, 3, 123, 126, 241,
  17, 0, 12, 206, 189, 108, 86]
key = b'HMV'

def ror(val, n):
    return ((val >> n) | (val << (8 - n))) & 0xFF

decoded = []
for i, c in enumerate(msg):
    c_rot = ror(c, i % 8)
    k = key[i % 3]
    decoded.append(chr(c_rot ^ k))

print(''.join(decoded))
